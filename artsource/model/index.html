    查看已否决的每像素算法 
    sourceFrame = resize(sourceAtlas.crop(frame × sourceHeight / N), targetRegion.w, targetRegion.h)
bodyCrop    = resize(targetBody.crop(targetRegion), targetRegion.w, targetRegion.h)
weight      = max(clamp((edgeSide - distanceToSide) / edgeSide, 0, 1))  // 外圈 1，内圈 0
output      = alphaAwareMix(sourceFrame, bodyCrop, weight)
outputAtlas = stack(output[0], output[1], … output[N - 1])
